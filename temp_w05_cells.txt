        self.value = value
        self.limit = limit

class Record:
    def __init__(self, label, value, limit):
        self.label = label
        self.value = value
        self.limit = limit

    def percent(self):
        """Return the value as a percentage of the limit."""
        return (self.value / self.limit) * 100

    def status(self):
        """Return OVER LIMIT, WARNING or OK for this record."""
        p = self.percent()
        if p >= 100:
            return "OVER LIMIT"
        elif p >= 90:
            return "WARNING"
        else:
            return "OK"

    def is_over(self):
        """Return True if the value is greater than the limit, otherwise False."""
        return self.value > self.limit


def load_records(path):
    """Read label,value,limit lines from a file and return a list of Record objects."""
    records = []
    with open(path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            label, value, limit = line.split(",")
            records.append(Record(label, float(value), float(limit)))
    return records


records = load_records("records_sample.txt")

over_count = 0
for r in records:
    if r.is_over():
        over_count += 1

print(f"{over_count} of {len(records)} records are OVER LIMIT")
